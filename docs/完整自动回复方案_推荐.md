## 📌 核心问题

当前的挑战：
1. ❌ 滑动距离和坐标的映射容易出错
2. ❌ 评论高度不固定，计算复杂
3. ❌ 网络延迟导致元素位置不稳定
4. ❌ 长时间运行容易出现累积误差

### 方案设计：**实时定位 + 动态点击**

```
┌─────────────────────────────────────────────────────────────┐
│  核心理念：永远不依赖计算，始终使用实时 UI 元素查询         │
```

## 🛠️ 实现代码框架

### 1. 关键字匹配模块

```python
class KeywordMatcher:
    def __init__(self, keywords):
        self.keywords = keywords  # 关键字列表
        self.matched = []

    def match_comments(self, comments):
        """对评论列表进行关键字匹配"""
        for comment in comments:
            text = comment['text']
            for keyword in self.keywords:
                if keyword in text:
                    self.matched.append({
                        'comment': comment,
                        'keyword': keyword,
                        'matched_time': datetime.now().isoformat()
                    })
                    break  # 每条评论只匹配一次

        return self.matched
```

### 2. 动态定位模块 (核心!)

```python
class DynamicLocator:
    def __init__(self, device):
        self.device = device

    def find_comment_by_text(self, target_text, timeout=5):
        """
        在当前屏幕上查找评论

        关键: 直接从 UI 查询，不使用缓存坐标！
        """
        start_time = time.time()

        while time.time() - start_time < timeout:
            # 获取当前的 UI 层级
            xml_str = self.device.device.dump_hierarchy()
            root = ET.fromstring(xml_str)

            # 在所有评论元素中搜索
            for element in root.iter():
                if element.get('resource-id') == COMMENT_ITEM:
                    text = element.get('text', '')

                    # 检查是否匹配
                    if target_text in text:
                        # 解析坐标
                        bounds = self._parse_bounds(element.get('bounds', ''))
                        if bounds:
                            return {
                                'found': True,
                                'text': text,
                                'bounds': bounds,
                                'center_x': (bounds[0] + bounds[2]) // 2,
                                'center_y': (bounds[1] + bounds[3]) // 2,
                            }

            time.sleep(0.5)

        return {'found': False}

    def scroll_to_find_comment(self, target_text, max_scrolls=5):
        """
        如果当前屏幕找不到，滑动寻找
        """
        for scroll_idx in range(max_scrolls):
            # 尝试找到
            result = self.find_comment_by_text(target_text, timeout=2)

            if result['found']:
                return result

            # 没找到，向上滑动
            print(f"评论不在当前屏幕，滑动寻找... ({scroll_idx + 1}/{max_scrolls})")
            self.device.swipe(540, 1944 + 100, 540, 1944 - 100, duration=500)
            time.sleep(1)

        return {'found': False}
```

### 3. 自动回复模块 (核心!)

```python
class AutoReplyEngine:
    def __init__(self, device, locator):
        self.device = device
        self.locator = locator

    def reply_to_comment(self, comment_text, reply_text):
        """
        自动回复一条评论

        关键步骤:
        1. 使用 DynamicLocator 找到评论的最新位置
        2. 点击评论
        3. 输入回复
        4. 发送
        """
        print(f"\n🎯 回复评论: {comment_text[:40]}...")

        # 步骤 1: 找到评论
        result = self.locator.scroll_to_find_comment(
            comment_text,
            max_scrolls=5
        )

        if not result['found']:
            print(f"✗ 找不到评论，已跳过")
            return {'success': False, 'reason': '评论未找到'}

        print(f"✓ 找到评论，坐标: ({result['center_x']}, {result['center_y']})")

        try:
            # 步骤 2: 点击评论
            print("  → 点击评论...")
            self.device.click(result['center_x'], result['center_y'])
            time.sleep(1)

            # 步骤 3: 输入回复
            print("  → 输入回复...")
            reply_input = self.device.device(
                resourceId=DouyinElementIds.COMMENT_INPUT
            )
            reply_input.click()
            self.device.input_text(reply_text)
            time.sleep(1)

            # 步骤 4: 发送
            print("  → 发送回复...")
            send_btn = self.device.device(
                resourceId=DouyinElementIds.SEND_TEXT_COMMENT
            )
            send_btn.click()
            time.sleep(2)

            print(f"✓ 回复已发送！")
            return {'success': True}

        except Exception as e:
            print(f"✗ 回复失败: {e}")
            return {'success': False, 'reason': str(e)}
```

### 4. 主控制器

```python
class AutoReplyController:
    def __init__(self, device):
        self.device = device
        self.locator = DynamicLocator(device)
        self.reply_engine = AutoReplyEngine(device, self.locator)

    def run(self, comments_file, keywords_file=None):
        """
        执行完整的自动回复流程
        """
        # 加载评论列表
        with open(comments_file, 'r', encoding='utf-8') as f:
            data = json.load(f)
            comments = data['comments']

        print(f"\n📋 加载了 {len(comments)} 条评论\n")

        # 提取关键字
        keywords = self._load_keywords(keywords_file)
        print(f"🔑 关键字: {keywords}\n")

        # 匹配评论
        matcher = KeywordMatcher(keywords)
        matched = matcher.match_comments(comments)

        print(f"✓ 匹配了 {len(matched)} 条评论\n")

        # 自动回复
        results = []
        for idx, match in enumerate(matched, 1):
            comment = match['comment']
            keyword = match['keyword']

            # 生成回复文本（可以根据关键字定制）
            reply_text = self._generate_reply(keyword)

            # 执行回复
            print(f"\n[{idx}/{len(matched)}]")
            result = self.reply_engine.reply_to_comment(
                comment['text'],
                reply_text
            )

            results.append({
                'comment': comment,
                'keyword': keyword,
                'result': result,
            })

            # 等待，避免被封
            time.sleep(3)

        # 显示统计
        success_count = sum(1 for r in results if r['result']['success'])
        print(f"\n✅ 成功回复: {success_count}/{len(matched)} 条")

        return results

    def _load_keywords(self, keywords_file=None):
        """加载关键字"""
        if keywords_file and Path(keywords_file).exists():
            with open(keywords_file, 'r', encoding='utf-8') as f:
                return json.load(f)
        else:
            # 默认关键字
            return ['点赞', '支持', '👍', '棒', '顶']

    def _generate_reply(self, keyword):
        """根据关键字生成回复文本"""
        replies = {
            '点赞': '感谢点赞！🙏',
            '支持': '感谢支持，继续加油！💪',
            '👍': '谢谢！💕',
            '棒': '感谢评价！😊',
            '顶': '感谢支持！🎉',
        }
        return replies.get(keyword, '感谢评论！')
```
